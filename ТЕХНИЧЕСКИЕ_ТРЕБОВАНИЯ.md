```java
@EventHandler
public void onEndermanLookAt(EntityTargetLivingEntityEvent event) {
    if (event.getEntity() instanceof Enderman) {
        if (event.getTarget() instanceof Player player) {
            // Агрессия на игроков без тыквы
            ItemStack helmet = player.getEquipment().getHelmet();
            if (helmet == null || helmet.getType() != Material.CARVED_PUMPKIN) {
                // Агрессия включена
                event.setCancelled(false);
            }
        }
    }
}

@EventHandler
public void onCrystalDestroy(EntityDamageByEntityEvent event) {
    if (event.getEntity() instanceof EnderCrystal) {
        // Спавн голема-стража
        spawnCrystalGuardian(event.getEntity().getLocation());
    }
}

private void spawnCrystalGuardian(Location loc) {
    IronGolem guardian = (IronGolem) world.spawnEntity(loc, EntityType.IRON_GOLEM);
    guardian.setCustomName("§6Страж Кристала");
    guardian.getAttribute(Attribute.GENERIC_MAX_HEALTH).setBaseValue(25.0);
    guardian.setHealth(25.0);
    
    // Уже агрессивен к ближайшему игроку
    Player nearest = findNearestPlayer(loc);
    if (nearest != null) {
        guardian.setTarget(nearest);
    }
}

@EventHandler
public void onAllCrystalsDestroyed(EnderCrystalDestroyEvent event) {
    // Проверка: все ли кристаллы уничтожены
    if (areAllCrystalsDestroyed()) {
        startPhantomWaves();
    }
}

private void startPhantomWaves() {
    new BukkitRunnable() {
        int waves = 5; // 15 секунд / 3 секунды = 5 волн
        
        @Override
        public void run() {
            if (waves <= 0) {
                cancel();
                return;
            }
            
            // Спавн 1-5 фантомов на каждого игрока
            for (Player player : world.getPlayers()) {
                int count = 1 + random.nextInt(5);
                for (int i = 0; i < count; i++) {
                    Location loc = player.getLocation().add(
                        random.nextInt(20) - 10,
                        10 + random.nextInt(10),
                        random.nextInt(20) - 10
                    );
                    Phantom phantom = (Phantom) world.spawnEntity(loc, EntityType.PHANTOM);
                    phantom.setTarget(player);
                }
            }
            
            waves--;
        }
    }.runTaskTimer(plugin, 0L, 60L); // Каждые 3 секунды
}
```

```java
@EventHandler
public void onChestOpen(PlayerInteractEvent event) {
    if (event.getClickedBlock().getType() == Material.CHEST) {
        Location chestLoc = event.getClickedBlock().getLocation();
        
        // Проверка: сундук в городе Края
        if (isInEndCity(chestLoc)) {
            // Случайный шанс спавна артефакта
            if (random.nextDouble() < 0.2) { // 20% шанс
                spawnArtifact(chestLoc);
            }
        }
    }
}

private void spawnArtifact(Location loc) {
    // Определить, какой артефакт ещё не найден
    int artifactNumber = getNextArtifactNumber();
    
    ItemStack artifact = itemManager.createStoryItem("end_artifact_" + artifactNumber);
    
    // Добавить в сундук
    Chest chest = (Chest) loc.getBlock().getState();
    chest.getInventory().addItem(artifact);
    
    // Диалог
    dialogManager.playDialogForAll("artifact." + artifactNumber + ".found");
}

@EventHandler
public void onArtifactPickup(PlayerPickupItemEvent event) {
    ItemStack item = event.getItem().getItemStack();
    
    if (isArtifact(item)) {
        int count = dataManager.getArtifactsCollected();
        dataManager.setArtifactsCollected(count + 1);
        
        if (count + 1 == 5) {
            // Все артефакты собраны!
            dialogManager.playDialogForAll("artifacts.all_found");
        }
    }
}
```

```java
@EventHandler
public void onArtifactPlaced(PlayerInteractEvent event) {
    Block block = event.getClickedBlock();
    
    // Проверка: постамент ритуала
    if (isRitualPedestal(block.getLocation())) {
        ItemStack item = event.getItem();
        
        if (isArtifact(item)) {
            // Разместить артефакт на постаменте
            placeArtifactOnPedestal(block.getLocation(), item);
            event.getPlayer().getInventory().remove(item);
            
            // Эффекты
            showPedestalActivation(block.getLocation(), item);
            
            // Проверить: все ли артефакты размещены
            if (areAllArtifactsPlaced()) {
                startFinalRitual();
            }
        }
    }
}

private void startFinalRitual() {
    // Диалоги
    dialogManager.playDialogForAll("final.ritual_start");
    
    // Эффекты на всех игроков
    for (Player player : world.getPlayers()) {
        // Левитация II на 60 секунд
        player.addPotionEffect(new PotionEffect(
            PotionEffectType.LEVITATION, 1200, 1
        ));
        
        // Слепота на 20 секунд
        player.addPotionEffect(new PotionEffect(
            PotionEffectType.BLINDNESS, 400, 255
        ));
    }
    
    // Партиклы END_ROD по кругу
    spawnEndRodCircle();
    
    // Создать портал в небе (100x100)
    Location portalLoc = getStructureCenter().add(0, 50, 0);
    createSkyPortal(portalLoc, 100);
    
    // Через 55 секунд - телепортация
    new BukkitRunnable() {
        @Override
        public void run() {
            teleportToDarkDimension();
        }
    }.runTaskLater(plugin, 1100L);
}

private void createSkyPortal(Location center, int size) {
    // Создать портал END_GATEWAY 100x100 в небе
    for (int x = -size/2; x < size/2; x++) {
        for (int z = -size/2; z < size/2; z++) {
            Location loc = center.clone().add(x, 0, z);
            loc.getBlock().setType(Material.END_GATEWAY);
        }
    }
}

private void teleportToDarkDimension() {
    // ЗАГЛУШКА для будущего обновления
    // Сейчас просто телепортируем обратно в оверворлд
    
    for (Player player : world.getPlayers()) {
        Location spawnLoc = player.getWorld().getSpawnLocation();
        player.teleport(spawnLoc);
        
        // Эффекты возвращения
        player.playSound(spawnLoc, Sound.BLOCK_PORTAL_TRAVEL, 2.0f, 1.0f);
        world.spawnParticle(Particle.PORTAL, spawnLoc, 1000, 1, 1, 1, 1);
        
        // Финальные диалоги
        dialogManager.playDialog(player, "final.complete");
        
        // Достижение
        giveAchievement(player, "savior_of_worlds");
    }
    
    // Убрать портал
    removePortal();
    
    // День наступает
    world.setTime(1000);
    world.setStorm(false);
}
```
